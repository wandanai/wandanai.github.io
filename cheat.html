import 'package:flutter/material.dart';

class Home extends StatefulWidget {
  const Home({super.key});

  @override
  State<Home> createState() => _HomeState();
}

class _HomeState extends State<Home> {
  final codeuser = TextEditingController();
  final nameuser = TextEditingController();
  final surnameuser = TextEditingController();
  final salaryuser = TextEditingController();
  final teluser = TextEditingController();

  @override
  void dispose() {
    codeuser.dispose();
    nameuser.dispose();
    surnameuser.dispose();
    salaryuser.dispose();
    teluser.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: SafeArea(
        child: Container(
          decoration: const BoxDecoration(
            gradient: RadialGradient(
              colors: [Colors.white, Colors.white],
              radius: 1.0,
            ),
          ),
          child: Center(
            child: SingleChildScrollView(
              padding: const EdgeInsets.all(16),
              child: Column(
                mainAxisSize: MainAxisSize.min,
                children: [
                  Image.asset('images/logo.jpg', width: 150),
                  const SizedBox(height: 16),
                  textbox(codeuser, 'code:'),
                  textbox(nameuser, 'name:'),
                  textbox(surnameuser, 'surname:'),
                  textbox(salaryuser, 'salary:'),
                  textbox(teluser, 'tel:'),
                  const SizedBox(height: 16),
                  showButton(),
                ],
              ),
            ),
          ),
        ),
      ),
    );
  }

  // ---- paste the snippets below here ----
}

Dialog

Future<void> showMessage(BuildContext context, String message) async {
  await showDialog(
    context: context,
    builder: (context) => SimpleDialog(
      title: Text(message),
      children: <Widget>[
        Row(
          mainAxisAlignment: MainAxisAlignment.end,
          children: <Widget>[
            TextButton(
              onPressed: () => Navigator.pop(context),
              child: const Text('ปิด', style: TextStyle(color: Colors.red)),
            ),
          ],
        ),
      ],
    ),
  );
}

Text box

Widget textbox(TextEditingController controller, String label) {
  return Padding(
    padding: const EdgeInsets.symmetric(vertical: 6),
    child: TextField(
      controller: controller,
      decoration: InputDecoration(
        labelText: label,
        labelStyle: const TextStyle(
          color: Colors.black,
          fontWeight: FontWeight.bold,
        ),
        helperText: 'ที่แสดงใสกล่อง',
      ),
    ),
  );
}

Buttons

Widget acceptbtn() {
  return ElevatedButton(
    onPressed: () {
      showMessage(
        context,
        "ยินดีต้อนรับ พนักงานทุกท่าน สู่ Application SCG \n"
        "ชื่อ-นามสกุล: ${nameuser.text} ${surnameuser.text}\n"
        "เบอร์โทร: ${teluser.text}",
      );
    },
    style: ElevatedButton.styleFrom(
      backgroundColor: Colors.green,
      foregroundColor: Colors.white,
    ),
    child: const Text('Login'),
  );
}

Widget cancelbtn() {
  return ElevatedButton(
    onPressed: cleartext,
    style: ElevatedButton.styleFrom(
      backgroundColor: Colors.red,
      foregroundColor: Colors.white,
    ),
    child: const Text('Cancel'),
  );
}

Widget showButton() {
  return Row(
    mainAxisSize: MainAxisSize.min,
    children: <Widget>[acceptbtn(), const SizedBox(width: 6.0), cancelbtn()],
  );
}

void cleartext() {
  codeuser.clear();
  nameuser.clear();
  surnameuser.clear();
  salaryuser.clear();
  teluser.clear();
}

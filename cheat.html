  Widget cancelbtn() {
    return ElevatedButton(
      onPressed: () {
        cleartext();
      },
      child: Text('Cancel'),
      style: ButtonStyle(
        backgroundColor: MaterialStateProperty.all<Color>(Colors.red),
      ),
    );
  }

  Widget showButton() {
    return Row(
      mainAxisSize: MainAxisSize.min,
      children: <Widget>[acceptbtn(), SizedBox(width: 6.0), cancelbtn()],
    );
  }

  void cleartext() {
    codeuser.clear();
    nameuser.clear();
    surnameuser.clear();
    salaryuser.clear();
    teluser.clear();
  }
}

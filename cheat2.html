button function

ปิดท้ายไฟล์ด้วย } ของ class

  Widget signinButton() {
    return ElevatedButton(
      onPressed: () {
        if (ctrlname.text.isEmpty ||
            ctrlsurname.text.isEmpty ||
            ctrlroom.text.isEmpty ||
            ctrlnum.text.isEmpty) {
          normalDialog(context, "กรุณากรอกข้อมูลให้ครบถ้วนด้วยครับ");
          return;
        }
        int? num = int.tryParse(ctrlnum.text);
        if (num == null || num <= 0) {
          normalDialog(context, "จำนวนวันต้องเป็นตัวเลขมากกว่า 0");
          return;
        }
        String room = ctrlroom.text.trim().toLowerCase();
        int rent;
        double discount, total;
        if (room == 'vip') {
          rent = num * 700;
        } else {
          rent = num * 500;
        }
        discount = rent * 5 / 100;
        total = rent - discount;
        normalDialog(
          context,
          'รายการเช่าห้องพักของท่านเรียบร้อย\n'
          'ชื่อ-นามสกุล: ${ctrlname.text} ${ctrlsurname.text}\n'
          'ห้องพัก: ${ctrlroom.text}\n'
          'ค่าเช่า: $rent\n'
          'ส่วนลด: $discount\n'
          'ค่าเช่าสุทธิ: $total',
        );
      },
      child: Text('Submit'),
      style: ButtonStyle(
        backgroundColor: MaterialStateProperty.all(Colors.green),
      ),
    );
  }

  Widget cancelButton() {
    return ElevatedButton(
      onPressed: () {
        clearText();
      },
      child: Text('Cancel'),
      style: ButtonStyle(
        backgroundColor: MaterialStateProperty.all(Colors.red),
      ),
    );
  }

  Widget showButton() {
    return Row(
      mainAxisSize: MainAxisSize.min,
      children: [signinButton(), SizedBox(width: 8.0), cancelButton()],
    );
  }
}
